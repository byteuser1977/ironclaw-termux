### 2.4 Worker (任务执行器)

#### 2.4.1 执行循环

```rust
pub async fn execute_job(&self, job: Job) -> Result<JobOutput> {
    let mut context = JobContext::new(job.id);
    let mut iteration = 0;

    loop {
        iteration += 1;

        // 1. 检查取消信号
        if self.is_cancelled(job.id).await {
            return Ok(JobOutput::Cancelled);
        }

        // 2. 构建LLM请求
        let request = self.build_request(&context).await?;

        // 3. 调用LLM
        let response = self.llm.complete(request).await?;

        // 4. 解析响应
        let reasoning = self.parse_response(response)?;

        // 5. 执行工具调用
        for tool_call in reasoning.tool_calls {
            let result = self.execute_tool(&tool_call, &context).await?;
            context.add_tool_result(tool_call.id, result);
        }

        // 6. 检查完成条件
        if reasoning.is_complete() {
            return Ok(JobOutput::Success(reasoning.final_answer));
        }

        // 7. 检查迭代限制
        if iteration >= self.max_iterations {
            return Ok(JobOutput::MaxIterationsReached);
        }

        // 8. 更新上下文
        context.update(reasoning);
    }
}
```

#### 2.4.2 工具调用执行

**执行流程**
```rust
pub async fn execute_tool(
    &self,
    tool_call: &ToolCall,
    context: &JobContext,
) -> Result<ToolOutput> {
    // 1. 获取工具定义
    let tool = self.tools.get(&tool_call.name)?;

    // 2. 验证参数
    tool.validate_params(&tool_call.arguments)?;

    // 3. 检查权限
    self.check_permissions(&tool, context)?;

    // 4. 执行工具
    let raw_output = tool.execute(tool_call.arguments.clone()).await?;

    // 5. 安全净化
    let sanitized = self.safety.sanitize_tool_output(
        &tool_call.name,
        &raw_output,
    )?;

    // 6. 记录成本
    self.record_cost(&tool_call.name, &sanitized);

    Ok(ToolOutput {
        raw: raw_output,
        sanitized,
        cost: sanitized.cost,
    })
}
```

```rust
pub fn sanitize(&self, content: &str) -> SanitizedOutput {
    let mut warnings = Vec::new();
    let mut sanitized = content.to_string();

    for pattern in &self.patterns {
        if let Some(mat) = pattern.regex.find(&sanitized) {
            warnings.push(InjectionWarning {

                severity: pattern.severity,
```

```rust
pub struct RateLimit {

    window: Duration,
    requests: VecDeque<DateTime<Utc>>,
}

impl RateLimiter {
    pub fn check(&mut self, tool_name: &str) -> Result<()> {
        let limit = self.limits.get(tool_name)?;

        let now = Utc::now();
        limit.requests.retain(|&t| now - t < limit.window);

        if limit.requests.len() >= limit.max_requests {

        }

        limit.requests.push_back(now);
        Ok(())
    }
}
```

**融合算法 (Reciprocal Rank Fusion)**
```rust
pub fn reciprocal_rank_fusion(
    fts_results: Vec<SearchResult>,
    vector_results: Vec<SearchResult>,
    k: usize,
) -> Vec<SearchResult> {
    let mut scores: HashMap<Uuid, f64> = HashMap::new();

    for (rank, result) in fts_results.iter().enumerate() {
        let score = 1.0 / (k + rank + 1) as f64;
        *scores.entry(result.id).or_insert(0.0) += score;
    }

    for (rank, result) in vector_results.iter().enumerate() {
        let score = 1.0 / (k + rank + 1) as f64;
        *scores.entry(result.id).or_insert(0.0) += score;
    }

    let mut results: Vec<_> = scores.into_iter().collect();
    results.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());

    results.into_iter()
        .map(|(id, _)| fts_results.iter().find(|r| r.id == id).unwrap().clone())
        .collect()
}
```

```rust
    overlap_size: usize,

impl Chunker {
    pub fn chunk(&self, document: &MemoryDocument) -> Vec<Chunk> {
        let mut chunks = Vec::new();
        let mut start = 0;

        while start < document.content.len() {
            let end = (start + self.max_chunk_size)
                .min(document.content.len());

            // 在分隔符处切分
            let chunk_end = self.find_best_split(&document.content, start, end);

            chunks.push(Chunk {
                document_id: document.id,
                chunk_index: chunks.len(),
                content: document.content[start..chunk_end].to_string(),
            });

            start = chunk_end - self.overlap_size;
        }

        chunks
    }
}
```

```rust
pub struct FailoverProvider {
    providers: Vec<Arc<dyn LlmProvider>>,
    cooldowns: Vec<Option<DateTime<Utc>>>,
    current_index: usize,
    config: FailoverConfig,
}

impl FailoverProvider {
    pub async fn complete(&self, request: CompletionRequest) -> Result<CompletionResponse> {
        for (i, provider) in self.providers.iter().enumerate() {
            if self.is_in_cooldown(i) {
                continue;
            }

            match provider.complete(request.clone()).await {
                Ok(response) => {
                    self.on_success(i);
                    return Ok(response);
                }
                Err(e) if e.is_transient() => {
                    self.on_failure(i);
                    continue;
                }
                Err(e) => return Err(e),
            }
        }

        Err(Error::AllProvidersFailed)
    }
}
```

```rust
    }
```

```rust
    let mut engine = wasmtime::Engine::default();
    let module = wasmtime::Module::new(&engine, module)?;

    // 3. 配置能力
    let mut linker = wasmtime::Linker::new(&engine);
    self.setup_capabilities(&mut linker)?;

    // 4. 实例化
    let instance = linker.instantiate(&module)?;

    // 5. 调用入口函数
    let func = instance.get_typed_func::<(i32, i32), i32>("run")?;
    let result = func.call(params_ptr, params_len)?;

    // 6. 读取输出
    let output = self.read_output(&instance, result)?;

    Ok(output)
}
```

```rust
fn calculate_importance(turn: &Turn, context: &Context) -> f32 {
    let mut score = 1.0;

    // 用户明确标记
    if turn.is_marked_important() {
        score *= 2.0;
    }

    // 包含关键决策

        score *= 1.5;

        score *= 1.2;
    }

    // 时间衰减

    score *= 0.95_f32.powf(age_days);

    score
}
```

### 4.2 混合搜索算法

```rust
fn reciprocal_rank_fusion(
    results: &[Vec<SearchResult>],
    k: usize,
) -> Vec<SearchResult> {
    let mut scores: HashMap<Uuid, f64> = HashMap::new();

    for result_list in results {
        for (rank, result) in result_list.iter().enumerate() {
            let score = 1.0 / (k + rank + 1) as f64;
            *scores.entry(result.id).or_insert(0.0) += score;
        }
    }

    let mut sorted: Vec<_> = scores.into_iter().collect();
    sorted.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap());

    sorted.into_iter()
        .map(|(id, _)| find_result_by_id(id, results))
        .flatten()
        .collect()
}
```
